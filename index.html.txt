      <div>
        <label>Zi:</label><br />
        <select id="daySelect">
          <option value="luni">Luni</option>

        <input type="text" id="taskInput" placeholder="ex: flotări, masă, învățat" style="width:100%;" />
      </div>
      <div>
        <button id="addTaskBtn">Adaugă</button>
      </div>

    <div class="card">
      <div class="row">
        <div style="flex:1;">
          <strong>Rutina pe zi</strong>
          <div class="small">Vezi toate activitățile pentru ziua selectată.</div>
        </div>
        <button id="aiSuggestBtn">Sugestie AI (demo)</button>
      </div>
      <div id="tasksGrid" class="grid"></div>
    </div>

    <div class="card">
      <strong>Recomandări AI (demo)</strong>
      <div class="small">Simulat local, fără backend. Poți conecta ulterior la un API real.</div>
      <textarea id="aiOutput" readonly></textarea>
    </div>
  </div>

  <script>
    const daySelect = document.getElementById('daySelect');
    const timeInput = document.getElementById('timeInput');
    const taskInput = document.getElementById('taskInput');
    const addTaskBtn = document.getElementById('addTaskBtn');
    const tasksGrid = document.getElementById('tasksGrid');
    const aiSuggestBtn = document.getElementById('aiSuggestBtn');
    const aiOutput = document.getElementById('aiOutput');

    const STORAGE_KEY = 'routine_planner_data';

    function loadData() {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : {};
    }

    function saveData(data) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    }

    function getDayTasks(day) {
      const data = loadData();
      return data[day] || [];
    }

    function setDayTasks(day, tasks) {
      const data = loadData();
      data[day] = tasks;
      saveData(data);
    }

    function renderTasks() {
      const day = daySelect.value;
      const tasks = getDayTasks(day).sort((a, b) => a.time.localeCompare(b.time));
      tasksGrid.innerHTML = '';
      if (tasks.length === 0) {
        tasksGrid.innerHTML = '<div class="small">Nicio activitate pentru ziua asta încă.</div>';
        return;
      }
      tasks.forEach((t, idx) => {
        const hourDiv = document.createElement('div');
        hourDiv.className = 'hour';
        hourDiv.textContent = t.time;

        const taskDiv = document.createElement('div');
        taskDiv.className = 'task';
        taskDiv.textContent = t.text;

        const delBtn = document.createElement('button');
        delBtn.textContent = 'Șterge';
        delBtn.style.marginTop = '4px';
        delBtn.onclick = () => {
          const current = getDayTasks(day);
          current.splice(idx, 1);
          setDayTasks(day, current);
          renderTasks();
        };

        taskDiv.appendChild(document.createElement('br'));
        taskDiv.appendChild(delBtn);

        tasksGrid.appendChild(hourDiv);
        tasksGrid.appendChild(taskDiv);
      });
    }

    addTaskBtn.onclick = () => {
      const day = daySelect.value;
      const time = timeInput.value;
      const text = taskInput.value.trim();
      if (!time || !text) {
        alert('Pune și ora și activitatea, frt 😄');
        return;
      }
      const tasks = getDayTasks(day);
      tasks.push({ time, text });
      setDayTasks(day, tasks);
      timeInput.value = '';
      taskInput.value = '';
      renderTasks();
    };

    daySelect.onchange = renderTasks;

    aiSuggestBtn.onclick = () => {
      const day = daySelect.value;
      const tasks = getDayTasks(day).sort((a, b) => a.time.localeCompare(b.time));
      if (tasks.length === 0) {
        aiOutput.value = 'Nu ai activități în ziua asta. Încearcă să adaugi: sport dimineața, învățat la prânz, relaxare seara.';
        return;
      }
      let msg = `Analiză rutină pentru ${day}:\n\n`;
      msg += `• Ai ${tasks.length} activități.\n`;
      const hasSport = tasks.some(t => /flotari|sport|sala|alergare/i.test(t.text));
      const hasStudy = tasks.some(t => /invata|scoala|curs/i.test(t.text));
      const hasBreak = tasks.some(t => /pauza|relax/i.test(t.text));

      if (!hasSport) msg += '- Nu văd sport. Adaugă măcar 15–20 minute de mișcare.\n';
      if (!hasStudy) msg += '- Nu văd timp de învățat. Pune un bloc de 1–2 ore.\n';
      if (!hasBreak) msg += '- Nu văd pauze. Pune pauze scurte între activități.\n';

      msg += '\nSugestie: încearcă să grupezi activitățile importante dimineața și să lași seara pentru relaxare.';
      aiOutput.value = msg;
    };

    renderTasks();
  </script>
